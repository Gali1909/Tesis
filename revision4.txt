  /// Suma las ultimas [n] salidas registradas del lanzador.
  Future<Map<String, dynamic>> ultimasSalidasLanzador(int lanzadorId, int n) async {
    final db = await database;
    final r = await db.rawQuery('''
      SELECT COUNT(*) AS salidas,
             COALESCE(SUM(CAST(ROUND(innings_lanzados_juego * 3) AS INTEGER)), 0) AS outs,
             COALESCE(SUM(picheos_lanzados), 0) AS picheos,
             COALESCE(SUM(carreras_permitidas), 0) AS cp,
             COALESCE(SUM(carreras_limpias), 0) AS cl,
             COALESCE(SUM(hits_permitidos), 0) AS h,
             COALESCE(SUM(ponches), 0) AS k,
             COALESCE(SUM(bases_por_bolas), 0) AS bb
      FROM (
        SELECT e.* FROM estadistica_juego_lanzador e
        JOIN juego j ON j.juego_id = e.juego_id
        WHERE e.lanzador_id = ?
        ORDER BY j.fecha DESC, j.juego_id DESC
        LIMIT ?
      )
    ''', [lanzadorId, n]);

    final m = r.first;
    final outs = m['outs'] as int;
    final cl = m['cl'] as int;
    final h = m['h'] as int;
    final k = m['k'] as int;
    final bb = m['bb'] as int;
    return {
      ...m,
      'ip': '${outs ~/ 3}.${outs % 3}',
      'era': outs == 0 ? 0.0 : cl * 27 / outs,
      'whip': outs == 0 ? 0.0 : (bb + h) * 3 / outs,
      'k9': outs == 0 ? 0.0 : k * 27 / outs,
    };
  }

class LanzadorPerfilHistorico {
  final int lanzadorId;
  final int juegosGanados;
  final int juegosPerdidos;
  final int aperturas;
  final int salvados;
  final int juegosLanzados;
  final double inningsLanzados;
  final int ponches;
  final int basesPorBolas;
  final int hitsPermitidos;
  final int carrerasLimpias;

  LanzadorPerfilHistorico({
    required this.lanzadorId,
    this.juegosGanados = 0,
    this.juegosPerdidos = 0,
    this.aperturas = 0,
    this.salvados = 0,
    this.juegosLanzados = 0,
    this.inningsLanzados = 0,
    this.ponches = 0,
    this.basesPorBolas = 0,
    this.hitsPermitidos = 0,
    this.carrerasLimpias = 0,
  });

  double? get efectividad {
    if (inningsLanzados == 0) return null;
    return (carrerasLimpias * 9) / inningsLanzados;
  }

  double? get whip {
    if (inningsLanzados == 0) return null;
    return (basesPorBolas + hitsPermitidos) / inningsLanzados;
  }

  }
