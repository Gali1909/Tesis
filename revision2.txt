/* Copia el primer caracter UTF-8 de s a out (para iniciales con acentos). */
static int primer_char(const char *s, char *out)
{
    unsigned char c = (unsigned char)s[0];
    int n = (c < 0x80) ? 1 : (c >> 5) == 0x6 ? 2 : (c >> 4) == 0xE ? 3 : (c >> 3) == 0x1E ? 4 : 1;
    for (int i = 0; i < n && s[i]; i++) out[i] = s[i];
    return n;
}

static void iniciales(const char *nombre, char *out, size_t cap)
{
    out[0] = '\0';
    if (cap < 9) return;
    while (*nombre == ' ') nombre++;
    if (*nombre == '\0') return;
    int n = primer_char(nombre, out);
    out[n] = '\0';
    const char *sp = strchr(nombre, ' ');
    if (sp) {
        while (*sp == ' ') sp++;
        if (*sp) {
            int m = primer_char(sp, out + n);
            out[n + m] = '\0';
        }
    }
}

static void aplicar(const vista_t *v, bool en, const perfil_lanzador_t *p)
{
    char buf[64];

    mostrar(v->avatar, true);
    mostrar(v->datos, true);
    mostrar(v->numero, true);
    for (int i = 0; i < 3; i++) mostrar(v->tab[i], true);
    mostrar(v->panel_stats, true);

    lv_label_set_text(v->nombre, p->nombre);

    iniciales(p->nombre, buf, sizeof(buf));
    lv_label_set_text(v->iniciales, buf);

    snprintf(buf, sizeof(buf), "#%u", p->numero);
    lv_label_set_text(v->numero, buf);

    const char *mano = en ? (p->mano ? "Left-handed pitcher (LHP)" : "Right-handed pitcher (RHP)")
                          : (p->mano ? "Lanzador zurdo (LHP)"      : "Lanzador derecho (RHP)");
    if (p->edad) snprintf(buf, sizeof(buf), en ? "%s  %u yrs" : "%s  %u a\xC3\xB1os", mano, p->edad);
    else         snprintf(buf, sizeof(buf), "%s", mano);
    lv_label_set_text(v->datos, buf);

    const uint16_t enteros[10] = { p->g, p->p, 0, p->j, p->a, p->sv, 0, p->so, 0, p->bb };
    for (int i = 0; i < 10; i++) {
        if (i == 2)      snprintf(buf, sizeof(buf), "%u.%02u", p->pcl_x100 / 100, p->pcl_x100 % 100);
        else if (i == 6) snprintf(buf, sizeof(buf), "%u.%u",   p->il_x10 / 10,    p->il_x10 % 10);
        else if (i == 8) snprintf(buf, sizeof(buf), "%u.%02u", p->whip_x100 / 100, p->whip_x100 % 100);
        else             snprintf(buf, sizeof(buf), "%u", enteros[i]);
        lv_label_set_text(v->stat[i], buf);
    }

    const char **nombres = en ? NOMBRE_TIPO_EN : NOMBRE_TIPO_ES;
    for (int i = 0; i < PERFIL_MAX_LANZ; i++) {
        bool hay = (i < p->n_lanz) && (p->lanz[i].tipo < N_TIPOS);
        mostrar(v->fila[i], hay);
        if (!hay) continue;
        const perfil_lanz_t *l = &p->lanz[i];
        lv_label_set_text(v->fila_nombre[i], nombres[l->tipo]);
        lv_slider_set_range(v->fila_slider[i], VEL_MIN_MPH, VEL_MAX_MPH);
        lv_slider_set_value(v->fila_slider[i], (l->vmin + l->vmax) / 2, LV_ANIM_OFF);
        if (l->vmax == 0)       snprintf(buf, sizeof(buf), "-");
        else if (l->vmin == l->vmax) snprintf(buf, sizeof(buf), "%u mph", l->vmax);
        else                    snprintf(buf, sizeof(buf), "%u-%u mph", l->vmin, l->vmax);
        lv_label_set_text(v->fila_vel[i], buf);
    }
}

    if (vista_en(&v)) aplicar(&v, true, p);

// perfil_lanzador_rx.c
#include <string.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_log.h"
#include "ui.h"
#include "uart_bridge.h"
#include "perfil_lanzador_ui.h"
#include "perfil_lanzador_rx.h"

static const char *TAG = "PERFIL_RX";

/* Trama 0x15 (payload, enteros de 16 bits little-endian):
 *  [0] numero  [1] mano (0=der,1=izq)  [2] edad
 *  [3..16]  g, p, j, a, sv, so, bb          (7 x uint16)
 *  [17..22] pcl*100, whip*100, il*10        (3 x uint16)
 *  [23] n_lanz (0..5)
 *  [24..38] 5 x { slot del catalogo de la app, vmin, vmax }
 *  [39] largo del nombre (<= 24)
 *  [40..] nombre UTF-8 (sin '\0')
 */
#define PERFIL_HDR_LEN 40

static portMUX_TYPE s_mux = portMUX_INITIALIZER_UNLOCKED;
static perfil_lanzador_t s_perfil;        /* ultimo perfil recibido */
static bool s_hay = false;                /* hay perfil cargado */
static bool s_dirty = false;              /* cambio pendiente de aplicar a la UI */

static bool s_estaba_en_pantalla = false; /* solo tarea LVGL */
static bool s_seleccionado = false;       /* solo tarea LVGL */

static uint16_t u16(const uint8_t *p) { return (uint16_t)(p[0] | (p[1] << 8)); }

/* ---------------- Lado UART (otra tarea) ---------------- */
void perfil_lanzador_rx_on_frame(uint8_t tipo, const uint8_t *p, uint8_t len)
{
    if (tipo == UART_MSG_QUITAR_PERFIL) {
        taskENTER_CRITICAL(&s_mux);
        s_hay = false;
        s_dirty = true;
        taskEXIT_CRITICAL(&s_mux);
        return;
    }
    if (tipo != UART_MSG_PERFIL_DATOS) return;

    if (len < PERFIL_HDR_LEN || p[39] > len - PERFIL_HDR_LEN) {
        ESP_LOGW(TAG, "Perfil 0x15 invalido (len=%d)", len);
        return;
    }

    perfil_lanzador_t t;
    memset(&t, 0, sizeof(t));
    t.numero = p[0];
    t.mano   = p[1];
    t.edad   = p[2];
    t.g = u16(p + 3);  t.p = u16(p + 5);  t.j = u16(p + 7);  t.a = u16(p + 9);
    t.sv = u16(p + 11); t.so = u16(p + 13); t.bb = u16(p + 15);
    t.pcl_x100 = u16(p + 17); t.whip_x100 = u16(p + 19); t.il_x10 = u16(p + 21);
    t.n_lanz = p[23] > PERFIL_MAX_LANZ ? PERFIL_MAX_LANZ : p[23];
    for (int i = 0; i < PERFIL_MAX_LANZ; i++) {
        t.lanz[i].tipo = p[24 + i * 3];
        t.lanz[i].vmin = p[25 + i * 3];
        t.lanz[i].vmax = p[26 + i * 3];
    }
    uint8_t nl = p[39];
    if (nl > sizeof(t.nombre) - 1) nl = sizeof(t.nombre) - 1;
    memcpy(t.nombre, p + PERFIL_HDR_LEN, nl);
    t.nombre[nl] = '\0';

    taskENTER_CRITICAL(&s_mux);
    s_perfil = t;
    s_hay = true;
    s_dirty = true;
    taskEXIT_CRITICAL(&s_mux);
    ESP_LOGI(TAG, "Perfil recibido: %s #%u", t.nombre, t.numero);
}

/* ---------------- Lado LVGL (tarea de LVGL) ---------------- */
static void copiar_estado(bool *hay, bool *dirty, perfil_lanzador_t *out)
{
    taskENTER_CRITICAL(&s_mux);
    *hay = s_hay;
    if (dirty) { *dirty = s_dirty; s_dirty = false; }
    if (out) *out = s_perfil;
    taskEXIT_CRITICAL(&s_mux);
}

static void click_lanzador_cb(lv_event_t *e)
{
    (void)e;
    bool hay;
    perfil_lanzador_t p;
    copiar_estado(&hay, NULL, &p);
    if (!hay) return;
    s_seleccionado = true;
    perfil_lanzador_ui_mostrar(&p);
}

static void tick_cb(lv_timer_t *t)
{
    (void)t;
    lv_obj_t *act = lv_screen_active();
    bool aqui = (act != NULL) &&
                ((ui_Perfil_lanzadores_es != NULL && act == ui_Perfil_lanzadores_es) ||
                 (ui_Perfil_lanzadores_EN != NULL && act == ui_Perfil_lanzadores_EN));
    if (!aqui) { s_estaba_en_pantalla = false; return; }

    bool hay, dirty;
    perfil_lanzador_t p;
    copiar_estado(&hay, &dirty, &p);

    if (!s_estaba_en_pantalla) {
        /* Recien entramos a la pantalla: ningun lanzador seleccionado. */
        s_estaba_en_pantalla = true;
        s_seleccionado = false;
        perfil_lanzador_ui_reset();
        perfil_lanzador_ui_lista(hay, p.nombre);
        perfil_lanzador_ui_set_click_cb(click_lanzador_cb);
        return;
    }

    if (dirty) {
        perfil_lanzador_ui_lista(hay, p.nombre);
        if (!hay) { s_seleccionado = false; perfil_lanzador_ui_reset(); }
        else if (s_seleccionado) perfil_lanzador_ui_mostrar(&p);
    }
}

void perfil_lanzador_rx_init(void)
{
    static bool s_iniciado = false;   /* seguro si se llama mas de una vez */
    if (s_iniciado) return;
    s_iniciado = true;
    uart_bridge_set_perfil_callback(perfil_lanzador_rx_on_frame);
    lv_timer_create(tick_cb, 200, NULL);
}

static const char *TAG = "UART_BRIDGE";

#define UART_BRIDGE_PORT     UART_NUM_1
// Los pines IO6/IO7 NO son los que realmente conecta el socket "Wireless
// Module" hacia el UART del modulo C6 (esto se penso originalmente por
// "numero con numero" sin verificar contra el datasheet). Alineando la
// columna del socket del P4 (IO27,IO8,IO7,IO6,3V3,GND,NC) con la "Right
// Interface" del datasheet del modulo (U1RXD,GPIO21,GPIO22,GPIO23,VDD3V3,
// GND,VBUS), los pines 3V3 y GND coinciden exactamente en la misma
// posicion (5ta y 6ta) - eso confirma que el socket se inserta espejado
// y que los pines reales de UART son IO27 (-> U1RXD del modulo) e IO28
// (<- U1TXD del modulo), no IO6/IO7.
#define UART_BRIDGE_TX_PIN   GPIO_NUM_27
#define UART_BRIDGE_RX_PIN   GPIO_NUM_28
#define UART_BRIDGE_BAUD     115200

#define FRAME_END     0x55
#define MAX_PAYLOAD   64

esp_err_t uart_bridge_init(void)
{
    uart_config_t config = {
        .baud_rate = UART_BRIDGE_BAUD,
        .data_bits = UART_DATA_8_BITS,
        .parity    = UART_PARITY_DISABLE,
        .stop_bits = UART_STOP_BITS_1,
        .flow_ctrl = UART_HW_FLOWCTRL_DISABLE,
        .source_clk = UART_SCLK_DEFAULT,
    };

    esp_err_t err = uart_driver_install(UART_BRIDGE_PORT, 1024, 1024, 0, NULL, 0);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Error instalando driver UART: %s", esp_err_to_name(err));
        return err;
    }

    err = uart_param_config(UART_BRIDGE_PORT, &config);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Error configurando UART: %s", esp_err_to_name(err));
        return err;
    }

    err = uart_set_pin(UART_BRIDGE_PORT, UART_BRIDGE_TX_PIN, UART_BRIDGE_RX_PIN,
                        UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Error asignando pines UART: %s", esp_err_to_name(err));
        return err;
    }

    s_initialized = true;
    ESP_LOGI(TAG, "UART bridge listo en TX=%d RX=%d @ %d baud",
             UART_BRIDGE_TX_PIN, UART_BRIDGE_RX_PIN, UART_BRIDGE_BAUD);

    /* Tarea de recepcion: lee lo que llega del modulo C6. Distingue dos cosas:
     *  - texto (por ejemplo "ACK_C6\n"): se muestra en el log como antes.
     *  - tramas [0xAA][tipo][len][payload][chk][0x55]: se validan y se
     *    entregan al callback registrado (p. ej. sincronizacion de idioma). */
    xTaskCreate(uart_bridge_rx_task, "uart_bridge_rx", 4096, NULL, 5, NULL);

    return ESP_OK;
}

static uint8_t calcular_checksum(uint8_t tipo, uint8_t len, const uint8_t *payload)
{
    uint8_t chk = tipo ^ len;
    for (uint8_t i = 0; i < len; i++) {
        chk ^= payload[i];
    }
    return chk;
}

typedef enum {
    RX_IDLE,
    RX_TIPO,
    RX_LEN,
    RX_PAYLOAD,
    RX_CHK,
    RX_END
} rx_state_t;

static void uart_bridge_rx_task(void *pvParameters)
{
    uint8_t buf[64];
    uint8_t payload[MAX_PAYLOAD];
    char line[96];
    size_t line_len = 0;

    rx_state_t st = RX_IDLE;
    uint8_t tipo = 0, len = 0, idx = 0;

    while (1) {
        int n = uart_read_bytes(UART_BRIDGE_PORT, buf, sizeof(buf), 100 / portTICK_PERIOD_MS);
        for (int i = 0; i < n; i++) {
            uint8_t b = buf[i];

            switch (st) {
            case RX_IDLE:
                if (b == FRAME_START) {
                    st = RX_TIPO;
                } else if (b == '\n' || b == '\r') {
                    if (line_len > 0) {
                        line[line_len] = '\0';
                        ESP_LOGI(TAG, "Recibido del C6 (%d bytes): %s", (int)line_len, line);
                        line_len = 0;
                    }
                } else if (b >= 0x20 && b < 0x7F && line_len < sizeof(line) - 1) {
                    line[line_len++] = (char)b;
                }
                break;

            case RX_TIPO:
                tipo = b;
                st = RX_LEN;
                break;

            case RX_LEN:
                len = b;
                idx = 0;
                if (len > MAX_PAYLOAD) {
                    ESP_LOGW(TAG, "Trama entrante con len invalido (%d), descartada", len);
                    st = RX_IDLE;
                } else {
                    st = (len == 0) ? RX_CHK : RX_PAYLOAD;
                }
                break;

            case RX_PAYLOAD:
                payload[idx++] = b;
                if (idx >= len) st = RX_CHK;
                break;

            case RX_CHK:
                if (b == calcular_checksum(tipo, len, payload)) {
                    st = RX_END;
                } else {
                    ESP_LOGW(TAG, "Trama entrante con checksum invalido (tipo=0x%02X)", tipo);
                    st = RX_IDLE;
                }
                break;

            case RX_END:
                if (b == FRAME_END) {
                    ESP_LOGI(TAG, "Trama recibida (tipo=0x%02X, %d bytes de payload)", tipo, len);
                    if (s_frame_cb) s_frame_cb(tipo, payload, len);
                    /* Perfil del lanzador (0x14 quitar / 0x15 datos): callback aparte,
                     * sin alterar lo que ya recibe s_frame_cb. */
                    if (s_perfil_cb && (tipo == UART_MSG_QUITAR_PERFIL || tipo == UART_MSG_PERFIL_DATOS)) {
                        s_perfil_cb(tipo, payload, len);
                    }
                } else {
                    ESP_LOGW(TAG, "Trama entrante sin byte de fin (tipo=0x%02X)", tipo);
                }
                st = RX_IDLE;
                break;
            }
        }
    }
}

esp_err_t uart_bridge_send_frame(uint8_t tipo, const uint8_t *payload, uint8_t len)
{
    if (!s_initialized) {
        ESP_LOGE(TAG, "uart_bridge_send_frame llamado antes de init");
        return ESP_ERR_INVALID_STATE;
    }
    if (len > MAX_PAYLOAD) {
        ESP_LOGE(TAG, "Payload demasiado grande: %d bytes", len);
        return ESP_ERR_INVALID_ARG;
    }

    /* Buffer temporal: start + tipo + len + payload + checksum + end */
    uint8_t frame[3 + MAX_PAYLOAD + 2];
    size_t idx = 0;

    frame[idx++] = FRAME_START;
    frame[idx++] = tipo;
    frame[idx++] = len;
    memcpy(&frame[idx], payload, len);
    idx += len;
    frame[idx++] = calcular_checksum(tipo, len, payload);
    frame[idx++] = FRAME_END;

    int written = uart_write_bytes(UART_BRIDGE_PORT, (const char *)frame, idx);
    if (written != (int)idx) {
        ESP_LOGE(TAG, "Error enviando trama: escritos %d de %d bytes", written, (int)idx);
        return ESP_FAIL;
    }

    ESP_LOGI(TAG, "Trama enviada (tipo=0x%02X, %d bytes de payload)", tipo, len);
    return ESP_OK;
}

esp_err_t uart_bridge_send_wifi(bool encender)
{
    uint8_t payload = encender ? 1 : 0;
    return uart_bridge_send_frame(UART_MSG_WIFI, &payload, 1);
}

esp_err_t uart_bridge_send_bluetooth(bool encender)
{
    uint8_t payload = encender ? 1 : 0;
    return uart_bridge_send_frame(UART_MSG_BLUETOOTH, &payload, 1);
}

  static Uint8List armar(int tipo, int seq, List<int> payload) {
    final len = payload.length;
    final b = BytesBuilder();
    b.add([0xAA, 0x55, tipo, seq & 0xFF, len & 0xFF, (len >> 8) & 0xFF]);
    b.add(payload);

    }
    b.addByte(x & 0xFF);
